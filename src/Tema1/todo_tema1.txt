// Importa BufferedReader: envuelve un Reader para leer texto de forma eficiente usando un buffer en memoria (permite leer lÃ­neas enteras)
import java.io.BufferedReader
// Importa InputStreamReader: actÃºa como puente traduciendo flujos de bytes (InputStream) a flujos de caracteres (Reader) segÃºn la codificaciÃ³n
import java.io.InputStreamReader


fun main() {
    // 1. ProcessBuilder("bash", "-c", "cd / && ls"):
    //    - "bash": Invoca al intÃ©rprete de comandos Bourne Again SHell.
    //    - "-c": Flag que indica a bash que lea y ejecute el comando contenido en el siguiente String.
    //    - "cd / && ls": Comando compuesto; primero cambia al directorio raÃ­z (/) y, solo si tuvo Ã©xito (operador lÃ³gico &&), ejecuta 'ls' para listar su contenido.
    // 2. .start(): Solicita al Sistema Operativo que cree un nuevo proceso hijo y comience su ejecuciÃ³n inmediatamente. Devuelve un objeto de tipo Process.
    val process = ProcessBuilder("bash","-c","cd / && ls").start()

    // process.inputStream: Aunque se llame "inputStream", representa la SALIDA estÃ¡ndar (stdout) del proceso hijo.
    // Se llama inputStream porque, desde el punto de vista de nuestro programa padre en Kotlin, los datos "entran" hacia nosotros.
    // Se envuelve en InputStreamReader (bytes -> caracteres) y luego en BufferedReader (caracteres -> lÃ­neas).
    val reader = BufferedReader(InputStreamReader(process.inputStream))

    // reader.lines(): Devuelve un Stream<String> de Java 8+ con todas las lÃ­neas que el proceso hijo va imprimiendo.
    // .forEach { println(it) }: Recorre el flujo e imprime por la consola de Kotlin cada lÃ­nea ('it') hasta que el proceso hijo cierra su salida.
    reader.lines().forEach { println(it) }
}

/*
Este archivo aborda la gestiÃ³n de errores en procesos hijos y demuestra la diferencia
fundamental entre un fallo al lanzar el ejecutable desde la JVM (IOException)

*/

/*
Puntos clave para estudio de 003.kt:

* Diferencia crÃ­tica de examen entre IOException y errorStream:
    1. Si haces ProcessBuilder("comando_inexistente").start() directamente:
       El sistema operativo le dice a la JVM que no existe ningÃºn archivo binario con ese nombre, y start() aborta lanzando una IOException en Kotlin. No llega a crearse ningÃºn objeto Process ni hay errorStream que leer.
    2. Si haces ProcessBuilder("bash", "-c", "comando_inexistente").start():
       El binario '/bin/bash' SÃ existe en el sistema operativo, por lo que el proceso arranca sin lanzar IOException. Cuando bash intenta buscar "comando_inexistente" y no lo encuentra, es el propio bash quien escribe el mensaje de error en su salida de error estÃ¡ndar (process.errorStream) y termina con cÃ³digo de salida 127.

* FunciÃ³n de redirectErrorStream(Boolean):
    - En false (valor por defecto): Mantiene separados los canales. La salida normal se lee en process.inputStream y los errores en process.errorStream.
    - En true: Fusiona ambos canales. Todos los mensajes (normales y de error) se leen juntos desde process.inputStream, y process.errorStream queda vacÃ­o.
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Comentario original del profesor:
// Lanzamos el comando a travÃ©s del intÃ©rprete porque,
// si se pasa directamente a ProcessBuilder, start()
// lanza una IOException (el ejecutable no existe)
// y no llega a haber ninguna salida de error que leer.
// En cambio, el intÃ©rprete sÃ­ arranca y es Ã©l quien escribe
// el error en su salida de error.


import java.io.BufferedReader
import java.io.InputStreamReader

fun main() {
    // Configuramos ProcessBuilder para que arranque 'bash' y este intente ejecutar "comando_inexistente".
    // Como el ejecutable 'bash' sÃ­ existe en el sistema operativo, pb.start() tendrÃ¡ Ã©xito.
    val pb = ProcessBuilder("bash", "-c", "comando_inexistente")

    // redirectErrorStream(false): Indica que NO queremos mezclar la salida de error (stderr) con la salida estÃ¡ndar (stdout).
    // De este modo, stdout irÃ¡ por process.inputStream y stderr irÃ¡ por process.errorStream.
    pb.redirectErrorStream(false)

    // Arranca el proceso hijo en el sistema operativo
    val process = pb.start()

    // Lector conectado a la salida estÃ¡ndar del hijo (descriptor de fichero 1 en Linux: stdout)
    val stdOut = BufferedReader(InputStreamReader(process.inputStream))

    // Lector conectado a la salida de error del hijo (descriptor de fichero 2 en Linux: stderr)
    val stdErr = BufferedReader(InputStreamReader(process.errorStream))

    // Intentamos leer la salida normal; como el comando fallÃ³, este flujo estarÃ¡ vacÃ­o (no imprimirÃ¡ lÃ­neas)
    println("Salida estÃ¡ndar:")
    stdOut.lines().forEach { println(it) }

    // Leemos la salida de error; aquÃ­ bash habrÃ¡ escrito: "bash: comando_inexistente: command not found"
    println("Salida de error:")
    stdErr.lines().forEach { println(it) }
}

/*

* Comportamiento de waitFor():
    Convierte la ejecuciÃ³n del subproceso en sÃ­ncrona respecto al hilo que lo llama.
    El hilo principal del padre se queda pausado (bloqueado) en la lÃ­nea process.waitFor() hasta que el proceso hijo termina en el sistema operativo.

    Si no pusiÃ©ramos waitFor(), "Proceso finalizado." se imprimirÃ­a inmediatamente despuÃ©s de arrancar el hijo, sin esperar a que termine.

* Trampa tÃ©cnica con "bash", "-c" y argumentos separados:
    En este cÃ³digo se escribiÃ³ ProcessBuilder("bash", "-c", "sleep", "2").
    En Bash, la opciÃ³n "-c" solo lee el PRIMER String siguiente ("sleep") como comando a ejecutar, y asigna los siguientes Strings ("2") a las variables posicionales ($0, $1...).
    Por tanto, bash ejecuta "sleep" sin argumentos, falla inmediatamente por "falta un operando" y no duerme los 2 segundos.
    Para que funcione correctamente con bash -c debe escribirse en un solo String: ProcessBuilder("bash", "-c", "sleep 2")
    o bien invocar directamente al binario sin bash: ProcessBuilder("sleep", "2").

// Define que este archivo pertenece al paquete Tema1
package Tema1

fun main() {
    // Mensaje impreso por el proceso padre antes de lanzar el proceso hijo
    println("Iniciando proceso...")

    // Crea y arranca el proceso hijo.
    // (Nota: Recuerda que en bash -c conviene unir "sleep 2" en un solo String para que no ignore el "2").
    val process = ProcessBuilder("bash","-c","sleep", "2").start()

    // process.waitFor(): Bloquea la ejecuciÃ³n del proceso padre en esta lÃ­nea
    // hasta que el proceso hijo haya terminado por completo en el sistema operativo.
    process.waitFor()

    // Esta lÃ­nea solo se alcanza y ejecuta una vez que el subproceso ha finalizado
    println("Proceso finalizado.")

/*
Este archivo combina la programaciÃ³n multihilo (kotlin.concurrent.thread) con la programaciÃ³n
multiproceso (ProcessBuilder) para lanzar varios procesos pesados o lentos en paralelo
y esperar a que todos terminen utilizando join().
*/

/*
Puntos clave para estudio de 006.kt:

* Â¿Por quÃ© envolver ProcessBuilder dentro de un hilo (thread)?
    Aunque ProcessBuilder.start() arranca el proceso de forma asÃ­ncrona, process.waitFor() es bloqueante.
    Si hiciÃ©ramos un bucle normal en el main llamando a start() y waitFor(), no se lanzarÃ­a el segundo proceso hasta que terminara el primero (ejecuciÃ³n secuencial: 1s + 2s + 3s = 6 segundos).
    Al meter start() + waitFor() dentro de un bloque thread { ... }, cada hilo secundario se queda bloqueado en su propio waitFor(), permitiendo que los 3 procesos se ejecuten simultÃ¡neamente en el Sistema Operativo (tiempo total = 3 segundos).

* Diferencia entre waitFor() y join():
    - process.waitFor() -> El hilo de Kotlin espera a que termine un PROCESO externo del Sistema Operativo.
    - hilo.join()       -> El hilo principal (main) espera a que termine otro HILO dentro de la misma JVM.
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Importa la funciÃ³n 'thread' de Kotlin, que crea e inicia automÃ¡ticamente un objeto Thread de Java
import kotlin.concurrent.thread

fun main() {
    // Lista inmutable con los segundos que debe durar cada proceso (como Strings)
    val comandos = listOf("1", "2", "3")

    // comandos.map { ... }: Transforma cada elemento de la lista ("1", "2", "3") en un hilo (Thread) ya en ejecuciÃ³n.
    // La variable 'hilos' resultante es de tipo List<Thread>.
    val hilos = comandos.map { segundos ->
        // Crea y arranca un nuevo hilo concurrente para gestionar este subproceso
        thread {
            // El hilo lanza su proceso correspondiente.
            // (Nota: igual que en 004.kt, en bash -c deberÃ­a escribirse ProcessBuilder("bash", "-c", "sleep $segundos") para que tome el argumento).
            val process = ProcessBuilder("bash","-c","sleep", segundos).start()

            // Bloquea ÃšNICAMENTE a este hilo secundario hasta que su proceso 'sleep' termine
            process.waitFor()

            // Informa por consola quÃ© proceso acaba de terminar
            println("Proceso sleep $segundos terminado")
        }
    }

    // Recorre la lista de los 3 hilos lanzados y llama a .join() en cada uno.
    // Esto obliga al hilo principal (main) a esperar hasta que los 3 hilos hayan acabado antes de finalizar el programa.
    hilos.forEach { it.join() }
}

=== ARCHIVO: 007.kt ===
/*
Este archivo implementa una tuberÃ­a (pipe) programÃ¡tica entre dos procesos hijos independientes
utilizando el mÃ©todo InputStream.transferTo(OutputStream) para conectar la salida del primero
con la entrada del segundo desde Kotlin.
*/

/*
Puntos clave para estudio de 007.kt:

* Funcionamiento de transferTo():
    Lee todos los bytes que escupe el primer proceso por su salida estÃ¡ndar (echo.inputStream) y los escribe directamente en la entrada estÃ¡ndar del segundo proceso (tr.outputStream).
    Equivale a ejecutar en la terminal de Linux: echo hola | tr a-z A-Z

* Importancia vital de cerrar el outputStream del proceso receptor (tr.outputStream.close()):
    Comandos filtro como 'tr', 'grep', 'wc' o 'sort' se quedan leyendo de su entrada estÃ¡ndar indefinidamente hasta que reciben la seÃ±al de fin de archivo (EOF - End Of File).
    Si olvidamos poner tr.outputStream.close(), el comando 'tr' creerÃ¡ que el padre aÃºn puede enviarle mÃ¡s texto, nunca terminarÃ¡, y nuestro programa se quedarÃ¡ colgado (deadlock) al intentar leer su salida en reader.lines().
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

import java.io.BufferedReader
import java.io.InputStreamReader

fun main() {
    // 1. Arranca el primer proceso hijo: ejecuta 'echo hola', que escribe "hola\n" en su stdout (echo.inputStream)
    val echo = ProcessBuilder("bash", "-c", "echo hola").start()

    // 2. Arranca el segundo proceso hijo: ejecuta 'tr a-z A-Z' (translate), que queda a la espera de recibir texto
    //    por su stdin (tr.outputStream) para convertir los caracteres de minÃºsculas (a-z) a mayÃºsculas (A-Z).
    val tr = ProcessBuilder("bash", "-c","tr a-z A-Z").start()

    // 3. Transfiere todos los bytes desde la salida estÃ¡ndar de 'echo' hacia la entrada estÃ¡ndar de 'tr'
    echo.inputStream.transferTo(tr.outputStream)

    // 4. IMPRESCINDIBLE: Cierra la entrada estÃ¡ndar de 'tr' para enviarle la seÃ±al EOF (Fin de Archivo)
    //    y que 'tr' procese los datos y termine su ejecuciÃ³n.
    tr.outputStream.close()

    // 5. Conectamos un BufferedReader a la salida estÃ¡ndar del segundo proceso ('tr.inputStream')
    val reader = BufferedReader(InputStreamReader(tr.inputStream))

    // 6. Leemos e imprimimos el resultado final transformado por 'tr': "HOLA"
    reader.lines().forEach { println(it) }
}

/*
Este archivo realiza un muestreo periÃ³dico (polling) de los procesos que mÃ¡s CPU consumen

y la sintaxis funcional idiomÃ¡tica de Kotlin para lectura de flujos.

/*
Puntos clave para estudio de 008.kt:

* Desglose del comando de Bash "ps -eo pid,comm,%cpu --sort=-%cpu | head -n 4":
    - ps -e: Selecciona todos los procesos activos del sistema.
    - -o pid,comm,%cpu: Indica el formato de salida mostrando solo 3 columnas (PID, nombre del comando y % de uso de CPU).
    - --sort=-%cpu: Ordena los resultados por la columna %cpu en orden descendente (el signo '-' indica de mayor a menor).
    - | head -n 4: EnvÃ­a la tabla por una tuberÃ­a a 'head' para cortar y mostrar Ãºnicamente las 4 primeras lÃ­neas (la lÃ­nea de tÃ­tulos + el Top 3 de procesos).

* Lectura idiomÃ¡tica en Kotlin:
    En lugar de escribir BufferedReader(InputStreamReader(top.inputStream)), Kotlin ofrece la funciÃ³n de extensiÃ³n .bufferedReader() directamente sobre cualquier InputStream.
    AdemÃ¡s, .forEach(::println) utiliza una referencia a funciÃ³n (::println), que es mÃ¡s concisa que escribir { println(it) }.
*/

// Nota: En el archivo original se omitiÃ³ 'package Tema1', por lo que si estÃ¡ en la carpeta Tema1
// el IDE puede sugerir aÃ±adir 'package Tema1' al principio del fichero.
fun main() {

    // repeat(3) { ... }: Ejecuta el bloque de cÃ³digo 3 veces seguidas.
    // En cada vuelta, la variable implÃ­cita 'it' vale 0, 1 y 2 respectivamente.
    repeat(3) {

        // Pausa el hilo actual durante 2000 milisegundos (2 segundos) entre cada muestra
        Thread.sleep(2000)

        // Lanza el subproceso en bash que obtiene el Top 3 de procesos con mayor consumo de CPU
        val top = ProcessBuilder("bash",
            "-c",
            "ps -eo pid,comm,%cpu --sort=-%cpu | head -n 4"
        ).start()

        // Imprime el encabezado de la muestra actual (sumando 1 a 'it' para que muestre 1, 2 y 3)
        println("=== Muestra ${it + 1} ===")

        // .bufferedReader(): ExtensiÃ³n de Kotlin que envuelve el InputStream en un BufferedReader (UTF-8 por defecto).
        // .lines(): Obtiene el Stream de lÃ­neas de texto.
        // .forEach(::println): Imprime cada lÃ­nea por consola usando referencia a mÃ©todo.
        top.inputStream.bufferedReader()
            .lines()
            .forEach(::println)
    }
}

    No podemos pasar directamente "hijo.jar" como primer argumento de ProcessBuilder porque un archivo .jar no es un binario nativo del sistema operativo (.exe o ELF).

fun main() {
    // Comentario original del profesor:
    // El proceso hijo es una nueva JVM que ejecuta hijo.jar
    // 1. ProcessBuilder("java", "-jar", "src/Tema1/hijo.jar"): Prepara el comando para lanzar java ejecutando el jar.
    // 2. .redirectErrorStream(true): Fusiona la salida de error (stderr) con la salida estÃ¡ndar (stdout) del hijo.
    // 3. .start(): Inicia el subproceso y devuelve el objeto Process.
    val process = ProcessBuilder("java", "-jar", "src/Tema1/hijo.jar")
        .redirectErrorStream(true)
        .start()

    // Obtiene un BufferedReader sobre la salida del hijo (process.inputStream),
    // recorre todas las lÃ­neas que el hijo imprima y las muestra en la consola del padre aÃ±adiendo la etiqueta "[hijo]: " delante.
    process.inputStream.bufferedReader().lines().forEach { println("[hijo]: $it") }

    // process.waitFor() bloquea al padre hasta que la JVM hija termine y devuelve su cÃ³digo de salida (0 si todo fue bien).
    println("El hijo terminÃ³ con cÃ³digo ${process.waitFor()}")
}

// Importa la clase File de java.io para crear y gestionar el archivo temporal en el sistema de ficheros
import java.io.File

fun main() {

    // ==========================================================
    // PASO 1: Crear un archivo temporal
    // ==========================================================

    // Crea un fichero vacÃ­o en la carpeta temporal del sistema con prefijo "practica" y extensiÃ³n ".txt"
    val archivoTemporal = File.createTempFile("practica",".txt")
    println("Archivo temporal creado:")
    // .absolutePath obtiene la ruta completa en texto (ej. /tmp/practica12345.txt) para pasÃ¡rsela luego a bash
    println(archivoTemporal.absolutePath)
    println()

    // ==========================================================
    // PASO 2: Escribir contenido usando printf desde bash
    // ==========================================================
    // printf enviarÃ¡ varias lÃ­neas al archivo temporal.
    //
    // > redirige la salida al archivo.
    //
    // Lanzamos bash ejecutando printf con 4 saltos de lÃ­nea (\n) y redirigiendo (>) el resultado al fichero temporal
    val escribir = ProcessBuilder(
        "bash",
        "-c",
        "printf 'LÃ­nea 1\nLÃ­nea 2\nLÃ­nea 3\nLÃ­nea 4\n' > ${archivoTemporal.absolutePath}"
    ).start()

    // Bloqueamos el programa padre hasta que el proceso 'escribir' haya volcado los datos y cerrado el archivo
    escribir.waitFor()

    // ==========================================================
    // PASO 3: Contar lÃ­neas usando wc -l
    // ==========================================================
    // Lanzamos el comando 'wc -l' (word count, opciÃ³n -l para contar lÃ­neas) inyectÃ¡ndole el archivo por stdin con '<'
    val contar = ProcessBuilder(
        "bash",
        "-c",
        "wc -l < ${archivoTemporal.absolutePath}"
    ).start()

    // Leemos toda la salida de 'wc -l' como un Ãºnico String con readText() y eliminamos espacios/saltos de lÃ­nea con trim()
    val numeroLineas = contar.inputStream
        .bufferedReader()
        .readText()
        .trim()

    // Esperamos a que el proceso 'contar' finalice completamente
    contar.waitFor()

    // Mostramos el total de lÃ­neas contadas (debe mostrar 4)
    println("NÃºmero de lÃ­neas: $numeroLineas")
    println()

    // ==========================================================
    // PASO 4: Leer contenido usando cat
    // ==========================================================
    // Lanzamos el comando 'cat' pasÃ¡ndole la ruta del archivo temporal para que vuelque su contenido por stdout
    val leer = ProcessBuilder(
        "bash",
        "-c",
        "cat ${archivoTemporal.absolutePath}"
    ).start()

    // Capturamos todo el texto devuelto por 'cat' en la variable 'contenido'
    val contenido = leer.inputStream
        .bufferedReader()
        .readText()

    // Esperamos a que el proceso 'leer' termine
    leer.waitFor()

    // ==========================================================
    // PASO 5: Mostrar contenido
    // ==========================================================
    println("Contenido del archivo:")
    println("----------------------")
    println(contenido)

    // ==========================================================
    // PASO 6: Eliminar archivo temporal
    // ==========================================================
    // Programa el borrado automÃ¡tico del fichero temporal justo cuando la mÃ¡quina virtual (JVM) finalice
    archivoTemporal.deleteOnExit()

/*
Este archivo muestra cÃ³mo sincronizar varios procesos independientes que intentan escribir
en un mismo archivo compartido (registro_compartido.txt) utilizando bloqueos de fichero
a nivel de Sistema Operativo mediante FileChannel y FileLock (API Java NIO).
*/

/*
Puntos clave para estudio de BloqueoDeFicheros.kt:

* Â¿Por quÃ© usar FileChannel.lock() en lugar de ReentrantLock o @Synchronized?
    ReentrantLock, Semaphore o @Synchronized solo sincronizan HILOS dentro de una misma JVM (un Ãºnico proceso).
    Si lanzamos dos programas independientes (dos procesos con distinta JVM) que escriben a la vez en el mismo fichero, necesitamos un mecanismo a nivel de Sistema Operativo: FileChannel.lock().

* Opciones de apertura (StandardOpenOption):
    - CREATE: Crea el fichero si aÃºn no existe.
    - WRITE: Abre el canal con permisos de escritura.
    - APPEND: Garantiza que cada escritura se sitÃºe al final del archivo sin machacar lo anterior.

* GestiÃ³n automÃ¡tica de recursos con .use { ... } en Kotlin:
    La funciÃ³n .use { ... } es el equivalente en Kotlin al try-with-resources de Java.
    Se aplica tanto al FileChannel como al FileLock devuelto por canal.lock().
    Garantiza que, al terminar el bloque (o incluso si ocurre una excepciÃ³n en medio de la escritura), el cerrojo (FileLock) se libere automÃ¡ticamente y el canal se cierre, evitando dejar el archivo bloqueado permanentemente.
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Importa FileChannel (Java NIO) para operar con canales de ficheros que soportan bloqueo y mapeo en memoria
import java.nio.channels.FileChannel
// Importa Path para representar rutas del sistema de archivos de forma moderna (NIO.2)
import java.nio.file.Path
// Importa todas las constantes de StandardOpenOption (CREATE, WRITE, APPEND, READ...)
import java.nio.file.StandardOpenOption.*

fun main() {
    // Obtiene el identificador de proceso (PID) del proceso actual en el sistema operativo para distinguirlo en el log
    val pid = ProcessHandle.current().pid()

    // Define la ruta relativa hacia el archivo de texto que compartirÃ¡n los procesos
    val fichero = Path.of("src/Tema1/registro_compartido.txt")

    // Abre el canal sobre el fichero (creÃ¡ndolo si no existe, en modo escritura y aÃ±adiendo al final).
    // El bloque .use { canal -> ... } cerrarÃ¡ el FileChannel automÃ¡ticamente al finalizar.
    FileChannel.open(fichero, CREATE, WRITE, APPEND).use { canal ->
        println("[$pid] Esperando el bloqueo del fichero...")

        // canal.lock() solicita al Sistema Operativo un bloqueo exclusivo sobre todo el archivo.
        // Si otro proceso tiene el bloqueo en este instante, esta llamada detiene (bloquea) al proceso actual hasta que el otro lo suelte.
        // El segundo .use { ... } asegura que el objeto FileLock se cierre (liberando el bloqueo) al salir de las llaves.
        canal.lock().use { // Bloquea hasta que ningÃºn otro proceso tenga el lock
            println("[$pid] Bloqueo obtenido, escribiendo...")

            // En Java NIO, FileChannel no escribe Strings directamente, sino objetos ByteBuffer.
            // Convertimos el String a un array de bytes (.toByteArray()) y lo envolvemos con ByteBuffer.wrap(...) para escribirlo.
            canal.write(java.nio.ByteBuffer.wrap("LÃ­nea escrita por el proceso $pid\n".toByteArray()))

            // Pausa de 3 segundos mientras mantenemos el bloqueo retenido.
            // Esto permite ejecutar una segunda instancia del programa desde el IDE al mismo tiempo y comprobar cÃ³mo la segunda espera.
            Thread.sleep(3000) // Simula una escritura larga
        } // El bloqueo se libera al salir del bloque

        println("[$pid] Bloqueo liberado")
    }
}

    1. Paso de parÃ¡metros al script: Se pasan como argumentos separados en el constructor: ProcessBuilder("ruta/script.sh", "arg1", "arg2"). Dentro del script bash se recibirÃ¡n como $1 y $2.
    2. .directory(File("...")): Cambia el directorio de trabajo (working directory) en el que se ejecutarÃ¡ el proceso hijo. Si no se especifica, el hijo hereda el directorio de trabajo del proceso padre (normalmente la raÃ­z del proyecto en IntelliJ).
    3. .redirectErrorStream(true): Hace que tanto la salida normal (stdout) como los errores (stderr) del script viajen juntos por process.inputStream.

* ExplicaciÃ³n de la expresiÃ³n idiomÃ¡tica while (reader.readLine().also { line = it } != null):
    En Java se suele escribir: while ((line = reader.readLine()) != null).
    En Kotlin, la asignaciÃ³n '=' no devuelve un valor (no es una expresiÃ³n), por lo que no puede meterse directamente dentro de la condiciÃ³n del while.

    2. .also { line = it } guarda ese valor leÃ­do ('it') en la variable 'line' y devuelve el mismo valor original.

package Tema1

// Importaciones para manejo de directorios (File), lectura de flujos (InputStreamReader, BufferedReader) y creaciÃ³n de procesos
import java.io.File
import java.io.InputStreamReader
import java.io.BufferedReader
import java.lang.ProcessBuilder


fun main() {
    // Crear un proceso para ejecutar un script externo indicando su ruta
    val processBuilder = ProcessBuilder("path/to/your/script.sh")
    /*
    Comentario original del profesor:
    Si queremos pasar algo como parÃ¡metro al script:
        val processBuilder = ProcessBuilder("path/to/your/script.sh", "arg1", "arg2")
     */
    // Establece la carpeta desde la cual se ejecutarÃ¡ el script (afecta a las rutas relativas que use el script por dentro)
    processBuilder.directory(File("/path/to/your/script/directory")) // Establecer el directorio de trabajo

    // Une el canal de errores (stderr) con el canal de salida estÃ¡ndar (stdout)
    processBuilder.redirectErrorStream(true) // Redirigir errores al flujo de salida

    // Bloque try-catch para capturar errores (por ejemplo, IOException si el script no existe o no tiene permisos de ejecuciÃ³n +x)
    try {
        // Arranca el script en el sistema operativo
        val process = processBuilder.start() // Iniciar el proceso

        // Crea el lector con buffer conectado a la salida del proceso hijo
        val reader = BufferedReader(InputStreamReader(process.inputStream))

        // Variable nullable donde iremos guardando cada lÃ­nea leÃ­da
        var line: String?

        // Lee lÃ­nea a lÃ­nea del proceso, la asigna a 'line' mediante .also y repite mientras no llegue al final del flujo (null)
        while (reader.readLine().also { line = it } != null) {
            println(line)
        }

        // Bloquea hasta que el script finalice por completo y guarda su cÃ³digo de retorno (0 = Ã©xito)
        val exitCode = process.waitFor()
        println("Proceso terminado con cÃ³digo de salida: $exitCode")
    } catch (e: Exception) {
        // Imprime la traza del error por consola si fallÃ³ el arranque o la ejecuciÃ³n del proceso
        e.printStackTrace()
    }
}

/*
Este archivo implementa el lado "Escritor" de un sistema de ComunicaciÃ³n Entre Procesos (IPC)
basado en Memoria Compartida (Shared Memory), proyectando un archivo en la memoria RAM
mediante MappedByteBuffer (canal.map).
*/

/*
Puntos clave para estudio de EscritorMemoriaCompartida.kt:

* Â¿QuÃ© es un archivo mapeado en memoria (Memory-Mapped File / MappedByteBuffer)?
    Mediante canal.map(FileChannel.MapMode.READ_WRITE, 0, 1024), el Sistema Operativo reserva una regiÃ³n de 1024 bytes de memoria RAM virtual directamente vinculada al fichero "memoria_compartida.dat".
    Cuando varios procesos mapean el mismo archivo, comparten directamente esas mismas pÃ¡ginas de memoria RAM fÃ­sica. Es el mecanismo de IPC mÃ¡s rÃ¡pido porque evita copiar datos continuamente entre el espacio de usuario y el kernel.

* Permisos obligatorios en FileChannel.open para MapMode.READ_WRITE:
    Para poder mapear en modo READ_WRITE, es OBLIGATORIO abrir el FileChannel incluyendo tanto READ como WRITE (ademÃ¡s de CREATE por si el archivo aÃºn no existe). Si solo pusiÃ©ramos WRITE, canal.map() lanzarÃ­a una NonReadableChannelException.

* DiseÃ±o del protocolo de datos en el buffer (Offset 0 y Offset 4):
    Como la zona de memoria mide siempre 1024 bytes fijos, el proceso lector necesita saber cuÃ¡ntos bytes reales ocupa nuestro mensaje para no leer basura o bytes vacÃ­os (ceros) hasta el byte 1024:
    - PosiciÃ³n (offset) 0: Escribimos un entero (Int) con la longitud del mensaje (memoria.putInt(0, mensaje.size)). Como un Int en Java/Kotlin ocupa exactamente 4 bytes (32 bits), ocupa las posiciones 0, 1, 2 y 3.
    - PosiciÃ³n (offset) 4: Justo a continuaciÃ³n del Int (en el byte Ã­ndice 4), escribimos los bytes del texto (memoria.put(4, mensaje)).
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Importaciones de Java NIO para canales de fichero, rutas y opciones de apertura
import java.nio.channels.FileChannel
import java.nio.file.Files
import java.nio.file.Path
import java.nio.file.StandardOpenOption.*

fun main() {
    // Ruta del archivo binario que servirÃ¡ de soporte para la memoria compartida entre procesos
    val fichero = Path.of("src/Tema1/memoria_compartida.dat")

    // Abre el canal con permisos de creaciÃ³n, lectura y escritura, cerrÃ¡ndolo automÃ¡ticamente al salir del bloque .use
    FileChannel.open(fichero, CREATE, READ, WRITE).use { canal ->
        // Mapear 1024 bytes del fichero en memoria (desde el byte 0 hasta el 1024) en modo Lectura/Escritura.
        // Devuelve un objeto MappedByteBuffer llamado 'memoria'.
        val memoria = canal.map(FileChannel.MapMode.READ_WRITE, 0, 1024)

        // Convierte el String del mensaje en un array de bytes (en codificaciÃ³n UTF-8 por defecto)
        val mensaje = "Hola desde la memoria compartida".toByteArray()

        // Escribe en los primeros 4 bytes (Ã­ndices 0, 1, 2 y 3) el nÃºmero de bytes que mide el mensaje
        memoria.putInt(0, mensaje.size)     // Primero la longitud del mensaje

        // Escribe a partir del byte Ã­ndice 4 el contenido real del array de bytes del mensaje
        memoria.put(4, mensaje)             // DespuÃ©s el contenido

        println("Mensaje escrito en la memoria compartida")
    }
}

=== ARCHIVO: LectorMemoriaCompartida.kt ===
/*
Este archivo implementa el lado "Lector" del sistema de Memoria Compartida (pareja de EscritorMemoriaCompartida.kt).
Accede a la misma regiÃ³n de 1024 bytes en modo de solo lectura (READ_ONLY), recupera primero el tamaÃ±o

*/

/*
Puntos clave para estudio de LectorMemoriaCompartida.kt:

* Apertura en modo solo lectura (READ y MapMode.READ_ONLY):
    Como este proceso Ãºnicamente necesita consultar los datos escritos por el escritor, abre el FileChannel solo con StandardOpenOption.READ y mapea la regiÃ³n con FileChannel.MapMode.READ_ONLY. Si intentara llamar a memoria.put(...) lanzarÃ­a una ReadOnlyBufferException.

* DescodificaciÃ³n siguiendo el mismo protocolo binario:
    1. memoria.getInt(0): Lee los 4 primeros bytes (posiciones 0..3) y los interpreta como un Int que indica la longitud exacta del mensaje en bytes.
    2. ByteArray(longitud): Crea un array de bytes en memoria del tamaÃ±o exacto que indicÃ³ la cabecera.
    3. memoria.get(4, bytes): Copia desde el Ã­ndice 4 de la memoria compartida tantos bytes como caben en nuestro array 'bytes'.
    4. String(bytes): Convierte los bytes leÃ­dos de nuevo a un String legible en UTF-8.
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Importaciones de Java NIO para abrir el canal y mapear el archivo en modo lectura
import java.nio.channels.FileChannel
import java.nio.file.Path
import java.nio.file.StandardOpenOption.READ

fun main() {
    // Apunta exactamente al mismo fichero que utilizÃ³ el proceso EscritorMemoriaCompartida
    val fichero = Path.of("src/Tema1/memoria_compartida.dat")

    // Abre el canal Ãºnicamente con permiso de lectura (READ) y garantiza su cierre al terminar con .use
    FileChannel.open(fichero, READ).use { canal ->
        // Mapea en memoria RAM la misma regiÃ³n de 1024 bytes (desde la posiciÃ³n 0) en modo solo lectura
        val memoria = canal.map(FileChannel.MapMode.READ_ONLY, 0, 1024)

        // Lee el entero (4 bytes) guardado en la posiciÃ³n 0 para saber cuÃ¡ntos bytes mide el mensaje escrito
        val longitud = memoria.getInt(0)

        // Reserva un array de bytes con el tamaÃ±o exacto del mensaje
        val bytes = ByteArray(longitud)

        // Extrae los bytes del mensaje empezando en la posiciÃ³n 4 (justo despuÃ©s de los 4 bytes del Int de longitud)
        memoria.get(4, bytes)

        // Convierte el ByteArray a String e imprime el mensaje recuperado por consola
        println("LeÃ­do de la memoria compartida: ${String(bytes)}")
    }
}

=== ARCHIVO: Monitores.kt ===
/*
Este archivo implementa el patrÃ³n de diseÃ±o de concurrencia "Monitor" utilizando
ReentrantLock y variables de condiciÃ³n (Condition) para coordinar tanto la exclusiÃ³n mutua
como la espera activa/notificaciÃ³n entre hilos.
*/

/*
Puntos clave para estudio de Monitores.kt:

- Â¿QuÃ© diferencia a un Monitor de un simple Mutex?
    Un Mutex solo impide que dos hilos entren a la vez (exclusiÃ³n mutua).
    Un Monitor, ademÃ¡s de exclusiÃ³n mutua (ReentrantLock), encapsula el estado del recurso (recursoDisponible) y permite que un hilo que ya entrÃ³ pero no puede continuar se duerma liberando la llave temporalmente (condition.await()) y sea despertado cuando otro hilo cambie el estado (condition.signal()).

- Pregunta clÃ¡sica de examen: Â¿Por quÃ© se usa `while (!recursoDisponible)` y NO `if (!recursoDisponible)` antes de `condition.await()`?
    1. Despertares espurios (Spurious Wakeups): El sistema operativo puede despertar un hilo en espera sin que nadie haya llamado a signal().
    2. Competencia entre varios hilos: Si otro hilo se cuela antes de que el hilo despertado recupere el lock, el recurso podrÃ­a volver a estar ocupado. El bucle `while` obliga a volver a comprobar la condiciÃ³n antes de seguir.
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Importa Condition: permite suspender hilos (await) y despertarlos (signal / signalAll) dentro de un ReentrantLock
import java.util.concurrent.locks.Condition
// Importa ReentrantLock: cerrojo de exclusiÃ³n mutua explÃ­cito mÃ¡s flexible que el bloque synchronized tradicional
import java.util.concurrent.locks.ReentrantLock
// Importa thread para crear hilos fÃ¡cilmente
import kotlin.concurrent.thread
// Importa la extensiÃ³n withLock, que adquiere el cerrojo al entrar al bloque y lo libera automÃ¡ticamente en un finally al salir
import kotlin.concurrent.withLock

/*
Comentario original de la profesora:
En este ejemplo, la clase Monitor encapsula un recurso compartido y
 proporciona mÃ©todos para acceder a Ã©l de manera segura.
 El primero que llama a accederRecurso() lo ocupa;
 el segundo se queda esperando en condition.await()
 (lo que libera temporalmente el lock)
 hasta que el primero llama a liberarRecurso()
 y le avisa con condition.signal().
 */

class Monitor {
    // Cerrojo interno que garantiza que solo un hilo pueda ejecutar cÃ³digo dentro de accederRecurso o liberarRecurso a la vez
    private val lock = ReentrantLock()
    // Variable de condiciÃ³n asociada a este cerrojo para gestionar la cola de hilos dormidos esperando el recurso
    private val condition: Condition = lock.newCondition()
    // Variable de estado (predicado) que indica si el recurso estÃ¡ libre (true) u ocupado (false)
    private var recursoDisponible = true // Al principio el recurso estÃ¡ libre

    fun accederRecurso() {
        // Adquiere el cerrojo antes de comprobar o modificar 'recursoDisponible'
        lock.withLock {
            // Mientras el recurso estÃ© ocupado por otro hilo, este hilo no puede avanzar
            while (!recursoDisponible) {
                // condition.await() hace 3 cosas de forma atÃ³mica:
                // 1. Libera temporalmente el 'lock' para que el otro hilo pueda entrar a 'liberarRecurso()'.
                // 2. Duerme al hilo actual hasta que alguien llame a condition.signal().
                // 3. Antes de devolver el control, vuelve a adquirir el 'lock'.
                condition.await()
            }
            // Cuando sale del while, tenemos el lock y sabemos seguro que el recurso estaba libre
            println("Accediendo al recurso compartido")
            // Marcamos el recurso como ocupado para que cualquier otro hilo que entre despuÃ©s tenga que esperar
            recursoDisponible = false
        } // Al salir de withLock, se libera el cerrojo fÃ­sico (pero recursoDisponible sigue en false)
    }

    fun liberarRecurso() {
        // Adquiere el cerrojo para modificar el estado de forma segura
        lock.withLock {
            println("Liberando el recurso compartido")
            // Cambia el estado de nuevo a libre
            recursoDisponible = true
            // Despierta a un hilo que estuviera dormido en condition.await() avisÃ¡ndole de que ya puede reintentar
            condition.signal()
        } // Libera el cerrojo para que el hilo reciÃ©n despertado pueda adquirirlo y salir de su await()
    }
}

fun main() {
    // Instancia Ãºnica del monitor que compartirÃ¡n ambos hilos
    val monitor = Monitor()

    // Crea una lista de 2 hilos que intentarÃ¡n usar el mismo monitor simultÃ¡neamente
    val tareas = (1..2).map {
        thread {
            // Solicita permiso al monitor; el segundo hilo quedarÃ¡ bloqueado aquÃ­ hasta que el primero libere
            monitor.accederRecurso()
            // Simula durante 1 segundo que estÃ¡ utilizando el recurso exclusivo
            Thread.sleep(1000)
            // Devuelve el recurso y despierta al siguiente hilo en espera
            monitor.liberarRecurso()
        }
    }
    // El hilo principal (main) espera a que ambos hilos terminen su ejecuciÃ³n antes de finalizar el programa
    tareas.forEach { it.join() }
}

/*

// Importa BufferedReader para leer texto lÃ­nea a lÃ­nea desde la salida del proceso hijo
import java.io.BufferedReader
// Importa InputStreamReader para convertir los bytes que salen del hijo en caracteres
import java.io.InputStreamReader
// Importa OutputStreamWriter para convertir nuestros Strings de Kotlin en bytes que enviar a la entrada del hijo
import java.io.OutputStreamWriter
// Importa ProcessBuilder para configurar y lanzar el proceso del sistema operativo
import java.lang.ProcessBuilder

fun main() {
    // Comentario original de la profesora:
    // Crear un proceso para ejecutar el comando 'grep "kotlin"'
    // Como no se le indica ningÃºn archivo a grep, este se queda escuchando por su entrada estÃ¡ndar (stdin)
    val processBuilder = ProcessBuilder("bash", "-c","grep kotlin")
    val process = processBuilder.start()

    // Comentario original de la profesora:
    // Escribir en el pipe de entrada del proceso
    // Conectamos un OutputStreamWriter al process.outputStream (que es el stdin del proceso grep)
    val writer = OutputStreamWriter(process.outputStream)
    // Enviamos la primera lÃ­nea (contiene la palabra "kotlin", por lo que grep la dejarÃ¡ pasar)
    writer.write("kotlin es un lenguaje de programaciÃ³n\n")
    // Enviamos la segunda lÃ­nea (no contiene "kotlin", por lo que grep la descartarÃ¡)
    writer.write("java es otro lenguaje de programaciÃ³n\n")

// Importa la funciÃ³n helper thread de Kotlin para lanzar hilos
import kotlin.concurrent.thread


/*
Comentario original de la profesora:
En este ejemplo, la clase Counter tiene un mÃ©t0do increment que incrementa
 un contador de manera segura utilizando la palabra clave @Synchronized.
 La prueba testConcurrentIncrement crea 100 hilos, cada uno incrementando
 el contador 1000 veces. Al final, se verifica que el valor del contador sea
 100000, asegurando que la concurrencia se maneje correctamente.
 */

class Counter {
    // Estado interno compartido por todos los hilos
    private var count = 0

    // @Synchronized bloquea el monitor del objeto Counter antes de entrar al mÃ©todo y lo libera al salir.
    // Impide que dos hilos ejecuten 'count++' simultÃ¡neamente (evita la condiciÃ³n de carrera).
    @Synchronized
    fun increment() {
        count++
    }

    // Devuelve el valor acumulado del contador
    fun getCount(): Int {
        return count
    }
}

class CounterTest {

    @Test
    fun testConcurrentIncrement() {
        // Instancia Ãºnica del contador que compartirÃ¡n los 100 hilos
        val counter = Counter()

        // Crea una lista de 100 hilos (Ã­ndices del 0 al 99) que arrancan inmediatamente en paralelo
        val threads = List(100) {
            thread {
                // Cada uno de los 100 hilos llama 1000 veces al mÃ©todo sincronizado increment()
                repeat(1000) {
                    counter.increment()
                }
            }
        }

        // El hilo del test espera a que los 100 hilos hayan terminado sus 1000 repeticiones cada uno
        threads.forEach { it.join() }

        // Si la sincronizaciÃ³n funcionÃ³ y no se perdiÃ³ ningÃºn incremento: 100 * 1000 = 100000
        assertEquals(100000, counter.getCount())
    }
}

    - En Segundo Plano: El padre lanza al hijo y NO lo espera inmediatamente, sino que continÃºa ejecutando su propio cÃ³digo en paralelo (el bucle `repeat(3)`).

- Â¿Por quÃ© se usa `ProcessBuilder.Redirect.DISCARD` en procesos en segundo plano?
    Equivale a hacer `> /dev/null 2>&1` en Linux.
    Si un proceso hijo en segundo plano escribe mucho texto y el padre no lo lee (porque estÃ¡ ocupado haciendo otras cosas), el buffer de la tuberÃ­a del sistema operativo se llena y el proceso hijo se queda congelado (bloqueado).     `process.onExit()` devuelve un `CompletableFuture<Process>` (promesa asÃ­ncrona introducida en Java 9).
    Con `.thenAccept { p -> ... }`, registramos una funciÃ³n (callback) que se ejecutarÃ¡ automÃ¡ticamente en un hilo separado en el instante exacto en que el hijo termine, sin detener al hilo principal del padre.
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

fun main() {
    // Configura un subproceso que tardarÃ¡ 5 segundos en terminar ("sleep 5")
    val process = ProcessBuilder("bash","-c","sleep 5")
        // Comentario original de la profesora:
        // Descarta todo lo que el hijo imprima por stdout para no llenar buffers de memoria
        .redirectOutput(ProcessBuilder.Redirect.DISCARD) // Descartar la salida
        // Comentario original de la profesora:
        // Descarta todo lo que el hijo imprima por stderr
        .redirectError(ProcessBuilder.Redirect.DISCARD)  // Descartar los errores
        // Inicia el proceso hijo en el sistema operativo y devuelve el control inmediatamente al padre
        .start()

    // Imprime el PID del proceso hijo reciÃ©n arrancado
    println("Proceso iniciado en segundo plano con PID: ${process.pid()}")

    // Comentario original de la profesora:
    // Registrar quÃ© hacer cuando termine, sin bloquear al padre
    // onExit() vigila al proceso en segundo plano; cuando pasen los 5 segundos del sleep,
    // ejecutarÃ¡ automÃ¡ticamente este bloque lambda imprimiendo su PID y su cÃ³digo de salida (exitValue())
    process.onExit().thenAccept { p ->
        println("El proceso ${p.pid()} ha terminado con cÃ³digo ${p.exitValue()}")
    }

    // Comentario original de la profesora:
    // Mientras tanto, el programa padre sigue trabajando
    // Este bucle dura 3 repeticiones x 3 segundos = 9 segundos en total.
    // Como el hijo solo dura 5 segundos, veremos el mensaje de "El proceso X ha terminado..."
    // intercalado por consola en mitad del trabajo del padre (entre la iteraciÃ³n 1 y la 2).
    repeat(3) { i ->
        println("El padre sigue trabajando... ($i)")
        Thread.sleep(3000)
    }

    // Comentario original de la profesora:
    // Si el trabajo del padre fuera mÃ¡s corto que el del hijo, la JVM se cerrarÃ­a antes de que saltara el onExit().
    // Este waitFor() final garantiza que main() no se cierre si el hijo aÃºn siguiera vivo.
    process.waitFor() // Solo para que el ejemplo no termine antes que el hijo
}

/*
Este archivo explica el uso de SemÃ¡foros (java.util.concurrent.Semaphore) para controlar
el acceso concurrente de mÃºltiples hilos a un recurso compartido mediante un sistema de permisos.
*/

/*
Puntos clave para estudio de Semaforo.kt:

- Diferencia de examen entre un Mutex (`ReentrantLock`) y un SemÃ¡foro (`Semaphore`):
    1. NÃºmero de hilos permitidos:
       - Un `Mutex` (`ReentrantLock`) solo deja pasar a **1 hilo** a la vez.
       - Un `Semaphore(N)` tiene un contador interno de permisos (`permits`). Permite que entren hasta **N hilos simultÃ¡neamente** a la secciÃ³n crÃ­tica.
       - Caso particular (SemÃ¡foro Binario): Si lo creas con `Semaphore(1)` (como en este archivo), solo hay 1 permiso disponible, por lo que se comporta en la prÃ¡ctica igual que un Mutex.
    2. Propiedad del bloqueo:
       - En un `Mutex`, solo el hilo que hizo `lock()` tiene derecho a hacer `unlock()`.
       - En un `Semaphore`, no existe concepto de "propietario": cualquier hilo podrÃ­a llamar a `release()` para incrementar los permisos.

- Funcionamiento interno de `acquire()` y `release()`:
    - `semaphore.acquire()`: Si el contador de permisos es `> 0`, resta 1 al contador y el hilo entra. Si el contador es `0`, el hilo se bloquea esperando hasta que alguien libere un permiso.
    - `semaphore.release()`: Suma 1 al contador de permisos y despierta a un hilo que estuviera esperando en `acquire()`.
    - Regla de oro: `semaphore.release()` SIEMPRE debe ir dentro de un bloque `finally` para que el permiso se devuelva aunque salte una excepciÃ³n dentro del `try`.
*/

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Importa la clase Semaphore del paquete de utilidades de concurrencia de Java
import java.util.concurrent.Semaphore
// Importa la funciÃ³n helper thread para crear hilos en Kotlin
import kotlin.concurrent.thread

fun main() {
    // Comentario original de la profesora:
    // Crea un semÃ¡foro con 1 Ãºnico permiso disponible (SemÃ¡foro binario = exclusiÃ³n mutua)
    // Si pusiÃ©ramos Semaphore(3), podrÃ­an entrar hasta 3 hilos a la vez.
    val semaphore = Semaphore(1) // SemÃ¡foro binario

    // Creamos e iniciamos 2 hilos concurrentes con identificadores 1 y 2
    val tareas = (1..2).map { id ->
        thread {
            // Solicita 1 permiso al semÃ¡foro:
            // - El primer hilo que llega ve el contador en 1 -> lo deja en 0 y entra al bloque try.
            // - El segundo hilo que llega ve el contador en 0 -> se queda bloqueado aquÃ­ esperando.
            semaphore.acquire()
            try {
                // --- INICIO DE LA SECCIÃ“N CRÃTICA ---
                println("Proceso $id: Accediendo al recurso compartido")
                // Simula el uso del recurso durante 1 segundo
                Thread.sleep(1000)
                println("Proceso $id: Liberando el recurso compartido")
                // --- FIN DE LA SECCIÃ“N CRÃTICA ---
            } finally {
                // Comentario original de la profesora:
                // Devuelve el permiso al semÃ¡foro (el contador vuelve a 1) y despierta al hilo que esperaba en acquire()
                semaphore.release() // Liberar siempre, aunque haya una excepciÃ³n
            }
        }
    }
    // El hilo principal espera a que ambos hilos terminen antes de finalizar el programa
    tareas.forEach { it.join() }
}

/*
Puntos clave para estudio de Sockets.kt:

- Diferencia fundamental entre `ServerSocket` y `Socket`:
    - `ServerSocket(9999)`: NO sirve para enviar ni recibir datos directamente. Es Ãºnicamente el "recepcionista" o "centralita" que abre el puerto TCP 9999 en nuestra tarjeta de red y se queda esperando a que llamen.
    - `val client: Socket = server.accept()`: El mÃ©todo `.accept()` es **bloqueante** (detiene el programa hasta que un cliente se conecta). - Orden de ejecuciÃ³n obligatorio en tus pruebas con IntelliJ IDEA:
    1. Primero debes dar a "Run" en **`Sockets.kt` (Servidor)**. VerÃ¡s que imprime `"Esperando conexiones..."` y se queda pausado en `server.accept()`.
    2. Sin parar el servidor, das a "Run" en **`Sockets02.kt` (Cliente)**. En ese instante el servidor se desbloquea, imprime `"Cliente conectado: /127.0.0.1"` y muestra `"Recibido del cliente: Hola"`.
    *(Si ejecutas el cliente antes que el servidor, fallarÃ¡ con `java.net.ConnectException: Connection refused` porque nadie estÃ¡ escuchando en el puerto 9999).*

// Define que este archivo pertenece al paquete Tema1
package Tema1

// Importaciones para leer flujos de texto (BufferedReader, InputStreamReader) y escribir (PrintWriter)
import java.io.BufferedReader
import java.io.InputStreamReader
import java.io.PrintWriter
// Importaciones de red: ServerSocket (para el servidor que escucha) y Socket (para la conexiÃ³n activa con el cliente)
import java.net.ServerSocket
import java.net.Socket

fun main() {
    // Comentario original de la profesora:
    // Iniciar un servidor de sockets
    // Registra y abre el puerto TCP 9999 en la mÃ¡quina local.
    // (Ojo: Si ejecutas dos veces el servidor sin cerrar el anterior, lanzarÃ¡ BindException: Address already in use).
    val server = ServerSocket(9999)
    println("Esperando conexiones...")

    // server.accept(): Bloquea la ejecuciÃ³n del servidor hasta que un proceso cliente haga Socket("localhost", 9999).
    // Cuando el cliente se conecta, devuelve un objeto Socket ('client') que representa el canal de comunicaciÃ³n con Ã©l.
    val client = server.accept()
    // client.inetAddress: Muestra la direcciÃ³n IP desde la que se ha conectado el cliente (ej. /127.0.0.1)
    println("Cliente conectado: ${client.inetAddress}")

    // Comentario original de la profesora:
    // Leer datos del cliente
    // Conectamos un BufferedReader al flujo de entrada del socket del cliente (lo que el cliente escribe por red)
    val reader = BufferedReader(InputStreamReader(client.inputStream))
    var line: String?
    // Lee lÃ­nea a lÃ­nea los mensajes enviados por el cliente hasta que el cliente cierra su socket (readLine devuelve null)
    while (reader.readLine().also { line = it } != null) {
        println("Recibido del cliente: $line")
    }

    // Comentario original de la profesora:
    // Cerrar conexiones
    // Es buena prÃ¡ctica cerrar siempre en orden inverso: primero los flujos, luego el socket del cliente y por Ãºltimo el ServerSocket
    reader.close()
    client.close()
    server.close()
}
